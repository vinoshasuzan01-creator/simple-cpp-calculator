#include <iostream>
using namespace std;
int main()
{
    float num1;
    float num2;
    char Operation;
    
    cout<< "Enter your first number:";
    cin>>num1;
    
    cout<< "Enter your second number:";
    cin>>num2;
    
    cout<< "Operation:";
    cin>>Operation;
    
    if(Operation=='+')
    cout<<num1 + num2;
    
    else if(Operation=='*')
    cout<<num1 * num2;
    
    else if(Operation=='/')
    cout<<num1 / num2;
    
    else if(Operation=='-')
    cout<<num1 - num2;
    
    else
    cout<<"Invalid.";
    
    return 0;
}
